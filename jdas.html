async function loadLeads() { const leadsContainer = document.getElementById('leadsBody'); try { // MUST BE A RELATIVE PATH const response = await fetch('/api/admin/leads', { credentials: 'include' }); if (response.ok) { const leads = await response.json(); leadsContainer.innerHTML = ''; leads.forEach(lead => { const row = ` ${lead.name} ${lead.company} ${lead.email} ${lead.service} ${lead.challenge} `; leadsContainer.innerHTML += row; }); } else if (response.status === 401) { // If the server says "Unauthorized", go back to login window.location.href = '/admin'; } } catch (error) { console.error("Dashboard error:", error); } }

async function loadLeads() { const leadsContainer = document.getElementById('leadsBody'); try { const response = await fetch('/api/admin/leads', { method: 'GET', credentials: 'include' // THIS IS THE FIX FOR THE BLANK PAGE }); if (response.ok) { const leads = await response.json(); leadsContainer.innerHTML = ''; leads.forEach(lead => { const row = ` ${lead.name} ${lead.company} ${lead.email} ${lead.service} ${lead.challenge} `; leadsContainer.innerHTML += row; }); } else if (response.status === 401) { window.location.href = '/admin'; } } catch (error) { console.error("Dashboard error:", error); } } document.addEventListener('DOMContentLoaded', loadLeads);
